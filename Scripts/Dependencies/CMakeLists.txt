function(keire_fribidi_generator target source)
    add_executable(${target} EXCLUDE_FROM_ALL
        "${KEIRE_FRIBIDI_SOURCE}/gen.tab/${source}"
        "${KEIRE_FRIBIDI_SOURCE}/gen.tab/packtab.c")
    target_include_directories(${target} PRIVATE
        "${KEIRE_FRIBIDI_SOURCE}/gen.tab"
        "${KEIRE_FRIBIDI_SOURCE}/lib")
    target_compile_definitions(${target} PRIVATE
        DONT_HAVE_FRIBIDI_CONFIG_H
        DONT_HAVE_FRIBIDI_UNICODE_VERSION_H
        HAVE_STDLIB_H
        HAVE_STRING_H
        HAVE_STRINGIZE)
endfunction()

function(keire_fribidi_table output target input0)
    set(arguments
        "-DKEIRE_PROGRAM=$<TARGET_FILE:${target}>"
        "-DKEIRE_OUTPUT=${fribidi_generated}/${output}"
        "-DKEIRE_ARG0=2"
        "-DKEIRE_ARG1=${KEIRE_FRIBIDI_SOURCE}/gen.tab/unidata/${input0}")
    set(dependencies ${target} "${KEIRE_FRIBIDI_SOURCE}/gen.tab/unidata/${input0}")
    if(ARGC GREATER 3)
        list(APPEND arguments "-DKEIRE_ARG2=${KEIRE_FRIBIDI_SOURCE}/gen.tab/unidata/${ARGV3}")
        list(APPEND dependencies "${KEIRE_FRIBIDI_SOURCE}/gen.tab/unidata/${ARGV3}")
        set(name_index 3)
    else()
        set(name_index 2)
    endif()
    list(APPEND arguments "-DKEIRE_ARG${name_index}=${target}")
    add_custom_command(OUTPUT "${fribidi_generated}/${output}"
        COMMAND ${CMAKE_COMMAND} ${arguments} -P "${fribidi_capture}"
        DEPENDS ${dependencies}
        VERBATIM)
endfunction()
