cmake_minimum_required(VERSION 3.20)
project(KeireDependencies LANGUAGES C CXX)

if(NOT DEFINED KEIRE_SDL_SOURCE OR NOT EXISTS "${KEIRE_SDL_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "KEIRE_SDL_SOURCE must point at the locked SDL source tree")
endif()
if(NOT DEFINED KEIRE_ASSIMP_SOURCE OR NOT EXISTS "${KEIRE_ASSIMP_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "KEIRE_ASSIMP_SOURCE must point at the locked Assimp source tree")
endif()
if(NOT DEFINED KEIRE_JOLT_SOURCE OR NOT EXISTS "${KEIRE_JOLT_SOURCE}/Build/CMakeLists.txt")
    message(FATAL_ERROR "KEIRE_JOLT_SOURCE must point at the locked Jolt source tree")
endif()
if(NOT DEFINED KEIRE_RECAST_SOURCE OR NOT EXISTS "${KEIRE_RECAST_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "KEIRE_RECAST_SOURCE must point at the locked Recast source tree")
endif()
if(NOT DEFINED KEIRE_MINIAUDIO_SOURCE OR NOT EXISTS "${KEIRE_MINIAUDIO_SOURCE}/miniaudio.h")
    message(FATAL_ERROR "KEIRE_MINIAUDIO_SOURCE must point at the locked miniaudio source tree")
endif()

add_subdirectory("${KEIRE_SDL_SOURCE}" SDL)
add_subdirectory("${KEIRE_ASSIMP_SOURCE}" Assimp)
add_subdirectory("${KEIRE_JOLT_SOURCE}/Build" Jolt)
add_subdirectory("${KEIRE_RECAST_SOURCE}" RecastNavigation)
add_subdirectory("${KEIRE_MINIAUDIO_SOURCE}" miniaudio)

add_library(KeireSDLDependency INTERFACE)
target_link_libraries(KeireSDLDependency INTERFACE SDL3::SDL3-static)

add_custom_target(keire-dependencies ALL)
add_dependencies(keire-dependencies SDL3-static assimp Jolt Recast Detour DetourCrowd DetourTileCache miniaudio)

file(TO_CMAKE_PATH "${KEIRE_JOLT_SOURCE}/LICENSE" KEIRE_JOLT_LICENSE)
file(TO_CMAKE_PATH "${KEIRE_RECAST_SOURCE}/License.txt" KEIRE_RECAST_LICENSE)
file(TO_CMAKE_PATH "${KEIRE_MINIAUDIO_SOURCE}/LICENSE" KEIRE_MINIAUDIO_LICENSE)
install(FILES "${KEIRE_JOLT_LICENSE}" DESTINATION share/licenses/keire RENAME Jolt-LICENSE.txt)
install(FILES "${KEIRE_RECAST_LICENSE}" DESTINATION share/licenses/keire RENAME Recast-LICENSE.txt)
install(FILES "${KEIRE_MINIAUDIO_LICENSE}" DESTINATION share/licenses/keire RENAME miniaudio-LICENSE.txt)
